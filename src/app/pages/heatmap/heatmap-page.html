onTimeRangeChange(range: HeatmapTimeRange) {
  this.isLoading.set(true);
  this.api.getHeatmap(range.interval, range.startDate, range.endDate)
    .subscribe(data => {
      this.heatmapData.set(data);
      this.isLoading.set(false);
    });
}
